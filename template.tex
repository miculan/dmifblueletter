%%
%%  COSA FA LA CLASSE
%%    * Disegna la carta intestata blu su TUTTE le pagine: logo e testi in alto,

%%    * Il testo occupa l'area fra intestazione e piè di pagina: parte a 113.4 bp
%%      (4 cm) dal bordo alto e finisce 113.4 bp prima del bordo basso.
%%    * Dalla seconda pagina compare "Page n" in blu, in alto a destra (con
%%      l'opzione italian la parola è tradotta da babel).
%%    * Il resto è il comportamento di `letter': data e destinatario a destra,
%%      oggetto, saluto, testo, formula di chiusura e firma.
%% =============================================================================


%% =============================================================================
%%  1. CLASSE E OPZIONI
%% =============================================================================
%%  Le opzioni si scrivono fra parentesi quadre, separate da virgole.
%%
%%  LINGUA DELLA LETTERA
%%    (nessuna)   Lettera in inglese: "Subject", data "Udine, November 20, 2026".
%%                È il comportamento predefinito.
%%    italian     Lettera in italiano: "Oggetto", data "Udine, lì 20 novembre 2026"
%%                (la classe carica da sola babel con l'italiano: NON caricare
%%                babel a mano). Serve il pacchetto babel-italian, presente in
%%                TeX Live e Overleaf.
%%    english     Equivale a non scrivere nulla; si può omettere. Se presente,
%%                LaTeX può segnalare "Unused global option(s): english": è
%%                innocuo.
%%
%%  LINGUA DELL'INTESTAZIONE (indipendente da quella della lettera)
%%    (nessuna)    Intestazione in italiano: "Dipartimento di scienze
%%                 Matematiche, Informatiche e Fisiche", "33100 Udine, Italia".
%%    englishhead  Intestazione in inglese: "Department of Mathematics, Computer
%%                 Science and Physics", "33100 Udine, Italy"; le etichette
%%                 opzionali del piè di pagina diventano "Person in charge of
%%                 the procedure" e "Prepared by".
%%
%%  STAMPA
%%    nosfondo    Non disegna la carta intestata (né immagine né testi): serve
%%                per stampare su carta intestata reale già stampata.
%%
%%  CORPO DEL TESTO (opzioni standard di LaTeX)
%%    10pt (predefinito) | 11pt | 12pt      dimensione del carattere
%%    Le altre opzioni standard di `letter' (fleqn, leqno, draft...) sono
%%    passate alla classe. Il formato è sempre A4 e non si cambia.
%%
%%  Esempi
%%    \documentclass[11pt]{dmifblueletter}                    inglese / italiano
%%    \documentclass[italian,11pt]{dmifblueletter}            italiano / italiano
%%    \documentclass[11pt,englishhead]{dmifblueletter}        inglese / inglese
%%    \documentclass[italian,11pt,englishhead]{dmifblueletter}  italiano / inglese
%%    \documentclass[11pt,nosfondo]{dmifblueletter}           su carta già intestata
%%
%%  COMPILARE: scegliere la combinazione voluta
\documentclass[italian,11pt]{dmifblueletter}

\begin{document}

%% =============================================================================
%%  6. LA LETTERA
%% =============================================================================
%%  Ogni lettera sta in un ambiente `letter'. L'argomento è il destinatario:
%%  la PRIMA riga è il nome, le righe successive (separate da \\ ) l'indirizzo.
%%  Data e destinatario sono stampati in alto, allineati a destra, in un blocco
%%  largo metà del testo.
%%
%%  Si possono mettere PIÙ lettere nello stesso file: basta ripetere l'ambiente
%%  (vedi esempio commentato in fondo). Ognuna ricomincia da pagina 1 e ha il
%%  proprio oggetto; nome, firma, email e piè di pagina sono comuni.

%% COMPILARE: destinatario
\begin{letter}{\fillin{Titolo e Nome del destinatario} \\
               \fillin{Ente / Struttura} \\
               \fillin{Indirizzo}}

%% Oggetto (facoltativo, ma va scritto DENTRO l'ambiente letter, prima di
%% \opening). Stampato come "SUBJECT: ..." (in italiano "OGGETTO: ...").
%% Si può terminare con il punto o meno. Se si omette, non appare nulla.
\subject{\fillin{oggetto della lettera}}

%% Saluto iniziale. È anche il punto dove la classe stampa data, destinatario e
%% oggetto.
\opening{Egregio \fillin{Sir or Madam},}

%% --- Corpo della lettera -----------------------------------------------------
%%  Testo normale: i capoversi si separano con una riga vuota. Se il testo supera
%%  la pagina continua sulla successiva, sotto la stessa carta intestata.
\fillin{Primo capoverso: motivo della lettera.}

\fillin{Secondo capoverso: sviluppo.}

\fillin{Ultimo capoverso: conclusione e disponibilità a fornire informazioni.}

%% Formula di chiusura e firma. La classe lascia sotto la formula uno spazio
%% (circa 1.3 cm) per la firma autografa o per l'immagine. Se si usa firma.png e
%% si vuole avvicinarla alla formula, aggiungere un passo negativo, ad esempio
%% \closing{Sincerely,\vspace{-1cm}}
\closing{Cordialmente,}

%% --- Dopo la firma (tutti facoltativi) ---------------------------------------
%%  \ps{...}    post scriptum
%%  \cc{...}    destinatari in copia conoscenza (più righe separate da \\ )
%%  \encl{...}  elenco degli allegati
% \ps{P.S. \fillin{testo del poscritto}}
% \cc{\fillin{Nome Cognome} \\ \fillin{Nome Cognome}}
% \encl{\fillin{elenco degli allegati}}

\end{letter}
